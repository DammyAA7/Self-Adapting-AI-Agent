import csv
import datetime
import os

def add_todo(todo_text, path=None):
    '''Adds a new todo item to the CSV file with auto-generated ID and timestamp.
    Returns True if successful, False otherwise.'''
    file_path = path if path else "/home/aifahim/PycharmProjects/Self-Adapting-AI-Agent/Self-Adapting-AI-Agent-Prototype-IV/todo.csv"
    
    # Validate input
    if not isinstance(todo_text, str) or not todo_text.strip():
        return False
    
    try:
        # Read existing todos to get next ID
        todos = []
        max_id = 0
        if os.path.exists(file_path):
            with open(file_path, 'r', newline='') as f:
                reader = csv.DictReader(f)
                for row in reader:
                    todos.append(row)
                    if 'id' in row and row['id'].isdigit():
                        max_id = max(max_id, int(row['id']))
        
        # Create new todo
        new_todo = {
            'id': str(max_id + 1),
            'todo': todo_text.strip(),
            'status': 'pending',
            'priority': 'medium',
            'created_at': datetime.datetime.now().isoformat()
        }
        todos.append(new_todo)
        
        # Write back to file
        fieldnames = ['id', 'todo', 'status', 'priority', 'created_at']
        with open(file_path, 'w', newline='') as f:
            writer = csv.DictWriter(f, fieldnames=fieldnames)
            writer.writeheader()
            writer.writerows(todos)
        
        return True
    except (IOError, OSError, PermissionError):
        return False
    except Exception:
        return False

- Validate inputs before processing
