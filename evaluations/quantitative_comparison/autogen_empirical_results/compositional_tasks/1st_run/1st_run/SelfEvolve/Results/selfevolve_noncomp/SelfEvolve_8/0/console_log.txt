```python
import csv
import json
import inspect
from typing import List, Dict, Any

def movie_api():
    """
    Returns MovieLens data in different formats based on caller context:
    - v1: list of titles
    - v2: dict of metadata per movie
    - v3: JSON-LD structured data
    The version is auto-detected by inspecting the call stack.
    """
    # Helper: Load the CSV once
    def load_movies() -> List[Dict[str, Any]]:
        with open('movie.csv', encoding='utf-8') as f:
            reader = csv.DictReader(f)
            movies = []
            for row in reader:
                row['movieId'] = int(row['movieId'])
                movies.append(row)
            return movies

    # Helper: Detect version from call stack
    def detect_version() -> str:
        stack = inspect.stack()
        for frame in stack:
            # Check for version identifier in function, module, or filename
            for attr in ('function', 'filename', 'module'):
                value = getattr(frame, attr, '')
                if isinstance(value, str):
                    if 'v1' in value:
                        return 'v1'
                    elif 'v2' in value:
                        return 'v2'
                    elif 'v3' in value:
                        return 'v3'
        # Default to v1 for backward compatibility
        return 'v1'

    version = detect_version()
    movies = load_movies()

    if version == 'v1':
        # Return a list of movie titles
        return [movie['title'] for movie in movies]
    elif version == 'v2':
        # Return a dict with metadata
        return [
            {
                'movieId': movie['movieId'],
                'title': movie['title'],
                'genres': movie['genres'].split('|')
            }
            for movie in movies
        ]
    elif version == 'v3':
        # Return JSON-LD structured data
        jsonld = []
        for movie in movies:
            entry = {
                "@context": "https://schema.org",
                "@type": "Movie",
                "name": movie['title'],
                "genre": movie['genres'].split('|'),
                "identifier": movie['movieId']
            }
            jsonld.append(entry)
        return jsonld
    else:
        # Unknown version, fallback to v1
        return [movie['title'] for movie in movies]

        stack = inspect.stack()

                    elif 'v2' in value:
                        return 'v2'
                    elif 'v3' in value:
                        return 'v3'
        # Default to v1 for backward compatibility
        return 'v1'

    version = detect_version()
    movies = load_movies()

    if version == 'v1':
        # Return a list of movie titles
        return [movie['title'] for movie in movies]
    elif version == 'v2':
