    if len(product) == 2 and len(product[0]) == 2:
        a, b = product[0]
        c, d = product[1]
        # Characteristic equation: λ^2 - (a+d)λ + (ad-bc) = 0
        trace = a + d
        det = a * d - b * c
        discriminant = trace**2 - 4 * det
        if discriminant >= 0:
            sqrt_disc = discriminant**0.5
            eig1 = (trace + sqrt_disc) / 2
            eig2 = (trace - sqrt_disc) / 2
            eigenvalues = [eig1, eig2]
        else:
            sqrt_disc = (-discriminant)**0.5
            real_part = trace / 2
            imag_part = sqrt_disc / 2
            eigenvalues = [complex(real_part, imag_part), complex(real_part, -imag_part)]
    else:
        eigenvalues = "Eigenvalue calculation only supported for 2x2 matrices without numpy."

    return {
        "product": product,
        "eigenvalues": eigenvalues
    }
