import math

def calculate_stock_volatility(prices):
    """
    Calculates the volatility (as a percentage) of a list of daily stock prices.
    Volatility is defined as the square root of the variance of daily returns, times 100.
    Returns 0.0 if fewer than 2 prices are provided.
    """
    if len(prices) < 2:
        return 0.0

    # Calculate daily returns
    returns = []
    for i in range(1, len(prices)):
        if prices[i-1] == 0:
            returns.append(0.0)
        else:
            returns.append((prices[i] - prices[i-1]) / prices[i-1])

    # Calculate mean of returns
    mean_return = sum(returns) / len(returns)

    # Calculate variance
    variance = sum((r - mean_return) ** 2 for r in returns) / len(returns)

    # Volatility as percentage
    volatility = math.sqrt(variance) * 100
    return volatility
