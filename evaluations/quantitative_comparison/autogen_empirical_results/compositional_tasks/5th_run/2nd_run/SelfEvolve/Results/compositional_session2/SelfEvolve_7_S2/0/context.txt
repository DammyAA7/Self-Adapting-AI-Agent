def parse_sensor_reading(sensor_string):
    """
    Parses a sensor string like 'T:25.5|H:60|TS:1234567890' and returns a dictionary with
    keys 'temperature', 'humidity', 'timestamp'. Values are converted to float (T, H), or int (TS). 
    If conversion fails, values are set to None.
    """
    result = {'temperature': None, 'humidity': None, 'timestamp': None}
    key_map = {'T': 'temperature', 'H': 'humidity', 'TS': 'timestamp'}
    convert_func = {'T': float, 'H': float, 'TS': int}
    
    parts = sensor_string.split('|')
    for part in parts:
        if ':' not in part:
            continue
        key, value = part.split(':', 1)
        key = key.strip()
        value = value.strip()
        if key in key_map:
            try:
                result[key_map[key]] = convert_func[key](value)
            except (ValueError, TypeError):
                result[key_map[key]] = None
    return result
