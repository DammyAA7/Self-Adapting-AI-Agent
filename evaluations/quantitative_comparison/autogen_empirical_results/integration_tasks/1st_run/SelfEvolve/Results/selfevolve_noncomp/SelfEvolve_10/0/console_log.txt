import csv
from collections import defaultdict

def performance_report(employee_id=None, report_type='summary'):
    filename = 'performance_reviews.csv'
    results = []
    with open(filename, mode='r', encoding='utf-8') as csvfile:
        reader = csv.DictReader(csvfile)
        for row in reader:
            # Convert overall_rating to float for further processing
            if 'overall_rating' in row:
                row['overall_rating'] = float(row['overall_rating'])
            results.append(row)

    if report_type == 'summary':
        if not employee_id:
            raise ValueError("employee_id must be provided for summary report.")
        for row in results:
            if row['employee_id'] == employee_id:
                return {
                    'employee_id': row['employee_id'],
                    'overall_rating': row['overall_rating'],
                    'goals_status': row['goals_status']
                }
        return None  # employee_id not found

    elif report_type == 'detailed':
        if not employee_id:
            raise ValueError("employee_id must be provided for detailed report.")
        for row in results:
            if row['employee_id'] == employee_id:
                # Convert numeric fields to appropriate types
                detailed_row = dict(row)
                for field in ['technical_score', 'communication_score', 'teamwork_score', 'leadership_score', 'overall_rating']:
                    if field in detailed_row:
                        if field == 'overall_rating':
                            detailed_row[field] = float(detailed_row[field])
                        else:
                            detailed_row[field] = int(detailed_row[field])
                return detailed_row
        return None

    elif report_type == 'department':
        if employee_id is not None:
            raise ValueError("employee_id must be None for department report.")
        dept_data = defaultdict(lambda: {'sum':0.0, 'count':0})
        for row in results:
            dept = row['department']
            dept_data[dept]['sum'] += row['overall_rating']
            dept_data[dept]['count'] += 1
        output = {}
        for dept, vals in dept_data.items():
            output[dept] = {
                'avg_rating': round(vals['sum'] / vals['count'], 2) if vals['count'] > 0 else 0.0,
                'count': vals['count']
            }
