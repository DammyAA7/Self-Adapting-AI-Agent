import csv

def performance_report(employee_id=None, report_type='summary'):
    # Read the CSV file
    with open('performance_reviews.csv', newline='', encoding='utf-8') as csvfile:
        reader = csv.DictReader(csvfile)
        rows = list(reader)

    if report_type == 'summary':
        # Find the row matching the employee_id
        for row in rows:
            if row['employee_id'] == employee_id:
                return {
                    'employee_id': row['employee_id'],
                    'overall_rating': float(row['overall_rating']),
                    'goals_status': row['goals_status']
                }
        return None  # Employee not found

    elif report_type == 'detailed':
        # Find the row matching the employee_id and return all columns with proper types
        for row in rows:
            if row['employee_id'] == employee_id:
                # Convert numeric columns to appropriate types
                detailed_row = {}
                for key, value in row.items():
                    if key in ['technical_score', 'communication_score', 'teamwork_score', 'leadership_score', 'overall_rating']:
                        detailed_row[key] = float(value)
                    else:
                        detailed_row[key] = value
                return detailed_row
        return None  # Employee not found

    elif report_type == 'department' and employee_id is None:
        # Group rows by department
        from collections import defaultdict
        dept_data = defaultdict(list)
        for row in rows:
            dept = row['department']
            dept_data[dept].append(float(row['overall_rating']))

        result = {}
        for dept, ratings in dept_data.items():
            avg_rating = sum(ratings) / len(ratings) if ratings else 0.0
            result[dept] = {
                'avg_rating': avg_rating,
                'count': len(ratings)
            }
        return result

    else:
        raise ValueError("Invalid parameters: employee_id and report_type combination is not supported.")
