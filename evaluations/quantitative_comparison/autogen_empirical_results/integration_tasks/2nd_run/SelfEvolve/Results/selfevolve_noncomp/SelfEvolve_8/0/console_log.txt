import csv
import json
import inspect
from typing import List, Dict, Any

def movie_api(query: str) -> Any:
    """
    Smart Movie API that returns data in different formats depending on the caller's context:
    - v1: Returns a list of movie titles matching the query.
    - v2: Returns a list of dicts with each movie's metadata.
    - v3: Returns a list of JSON-LD structured data for each movie.

    The API inspects the call stack to auto-detect which version to use.
    """
    # Helper to load movies from the CSV
    def load_movies() -> List[Dict[str, Any]]:
        movies = []
        with open('movie.csv', newline='', encoding='utf-8') as csvfile:
            reader = csv.DictReader(csvfile)
            for row in reader:
                # Clean up genres splitting
                row['genres'] = row['genres'].split('|')
                movies.append(row)
        return movies

    # Determine API version from the call stack
    stack = inspect.stack()
    version = 'v1'  # Default version
    for frame in stack:
        fname = frame.function.lower()
        if 'v2' in fname:
            version = 'v2'
            break
        elif 'v3' in fname:
            version = 'v3'
            break

    movies = load_movies()
    # Filter matching movies by query (case-insensitive substring match in title)
    results = [m for m in movies if query.lower() in m['title'].lower()]

    if version == 'v1':
        # Return list of titles
        return [m['title'] for m in results]
    elif version == 'v2':
        # Return list of dicts with metadata
        return [
            {
                "movieId": int(m["movieId"]),
                "title": m["title"],
                "genres": m["genres"]
            } for m in results
        ]
    elif version == 'v3':
        # Return list of JSON-LD structured data
        return [
            {
                "@context": "https://schema.org",
                "@type": "Movie",
                "name": m["title"],
                "genre": m["genres"],
                "identifier": m["movieId"]
            } for m in results
        ]
    else:
        # Fallback: just return titles
        return [m['title'] for m in results]
