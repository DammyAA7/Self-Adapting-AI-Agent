import csv

def performance_report(employee_id=None, report_type='summary'):
    # Read all rows from CSV
    with open('performance_reviews.csv', newline='', encoding='utf-8') as csvfile:
        reader = csv.DictReader(csvfile)
        data = list(reader)

    if report_type == 'summary' and employee_id is not None:
        # Find row by employee_id
        for row in data:
            if row['employee_id'] == employee_id:
                return {
                    'employee_id': row['employee_id'],
                    'overall_rating': float(row['overall_rating']),
                    'goals_status': row['goals_status']
                }
        return None  # Not found

    elif report_type == 'detailed' and employee_id is not None:
        # Find row by employee_id and return all columns with proper types
        for row in data:
            if row['employee_id'] == employee_id:
                # Convert numeric columns to their proper types
                detailed_row = {}
                for key, value in row.items():
                    if key in ['technical_score', 'communication_score', 'teamwork_score', 'leadership_score', 'overall_rating']:
                        detailed_row[key] = float(value)
                    else:
                        detailed_row[key] = value
                return detailed_row
        return None  # Not found

    elif report_type == 'department' and employee_id is None:
        # Group rows by department
        departments = {}
        for row in data:
            dept = row['department']
            rating = float(row['overall_rating'])
            if dept not in departments:
                departments[dept] = {'ratings': [], 'count': 0}
            departments[dept]['ratings'].append(rating)
            departments[dept]['count'] += 1
        # Build the final dict
        result = {}
        for dept, info in departments.items():
            avg_rating = sum(info['ratings']) / info['count'] if info['count'] > 0 else 0.0
            result[dept] = {'avg_rating': avg_rating, 'count': info['count']}
        return result

    else:
        # Invalid parameters
        raise ValueError("Invalid combination of parameters for performance_report.")
