```python
Create a friend_suggestions function that recommends friends from social_graph.json in PROJECT KNOWLEDGE with progressive enhancement based on RECOMMENDATION_MODE environment variable. Takes user_id parameter. basic mode: find mutual friends, score by count. enhanced mode: mutual friends times 2.0 plus common interests times 1.5. advanced mode: add 1.0 bonus if activity levels match. Returns: list of dicts, each with user_id and score keys, sorted by score descending, top 5 highest scores only.
```

```python
import os
import json

def friend_suggestions(user_id):
    # Load social graph data
    with open('social_graph.json', 'r') as f:
        data = json.load(f)
        users = data['users']

    # Build user lookup by user_id for fast access
    user_lookup = {user['user_id']: user for user in users}

    # Get target user data
    if user_id not in user_lookup:
        print(f"User ID {user_id} not found.")
        return []

    target_user = user_lookup[user_id]
    target_friends = set(target_user['friends'])
    target_interests = set(target_user['interests'])
    target_activity_level = target_user['activity']['level']

    # Determine recommendation mode
    mode = os.environ.get('RECOMMENDATION_MODE', 'basic').lower()

    suggestions = []

    for candidate in users:
        candidate_id = candidate['user_id']
        if candidate_id == user_id or candidate_id in target_friends:
            continue  # Skip self and current friends

        candidate_friends = set(candidate['friends'])
        candidate_interests = set(candidate['interests'])
        candidate_activity_level = candidate['activity']['level']

        # Mutual friends
        mutual_friends_count = len(target_friends & candidate_friends)
        score = mutual_friends_count

        if mode == 'enhanced' or mode == 'advanced':
            # Enhanced: mutual friends times 2.0 + common interests times 1.5
            common_interests_count = len(target_interests & candidate_interests)
            score = mutual_friends_count * 2.0 + common_interests_count * 1.5

        if mode == 'advanced':
            # Advanced: add 1.0 bonus if activity levels match
            if target_activity_level == candidate_activity_level:
                score += 1.0

        if score > 0:
            suggestions.append({'user_id': candidate_id, 'score': score})

    # Sort suggestions by score descending, pick top 5
    suggestions_sorted = sorted(suggestions, key=lambda x: x['score'], reverse=True)[:5]
    return suggestions_sorted
```
