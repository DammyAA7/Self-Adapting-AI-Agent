import csv
from collections import defaultdict

def performance_report(employee_id=None, report_type='summary'):
    filename = 'performance_reviews.csv'
    with open(filename, 'r', encoding='utf-8') as f:
        reader = csv.DictReader(f)
        rows = list(reader)

    if report_type == 'summary':
        if employee_id is None:
            raise ValueError("employee_id must be provided for summary report.")
        for row in rows:
            if row['employee_id'] == employee_id:
                return {
                    'employee_id': row['employee_id'],
                    'overall_rating': float(row['overall_rating']),
                    'goals_status': row['goals_status']
                }
        return None  # Not found

    elif report_type == 'detailed':
        if employee_id is None:
            raise ValueError("employee_id must be provided for detailed report.")
        for row in rows:
            if row['employee_id'] == employee_id:
                # Convert appropriate columns to float/int
                detailed_row = dict(row)
                float_cols = ['technical_score', 'communication_score', 'teamwork_score', 'leadership_score', 'overall_rating']
                for col in float_cols:
                    detailed_row[col] = float(detailed_row[col])
                return detailed_row
        return None  # Not found

    elif report_type == 'department':
        if employee_id is not None:
            raise ValueError("employee_id should be None for department report.")
        dept_stats = defaultdict(lambda: {'total_rating': 0.0, 'count': 0})
        for row in rows:
            dept = row['department']
            rating = float(row['overall_rating'])
            dept_stats[dept]['total_rating'] += rating
            dept_stats[dept]['count'] += 1
        result = {}
        for dept, stats in dept_stats.items():
            avg_rating = stats['total_rating'] / stats['count'] if stats['count'] > 0 else 0.0
            result[dept] = {'avg_rating': avg_rating, 'count': stats['count']}
        return result

    else:
        raise ValueError("Invalid report_type. Must be 'summary', 'detailed', or 'department'.")

        rows = list(reader)

    if report_type == 'summary':
        if employee_id is None:

            if row['employee_id'] == employee_id:

                for col in float_cols:
                    detailed_row[col] = float(detailed_row[col])
                return detailed_row
        return None
