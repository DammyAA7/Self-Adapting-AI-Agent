# matrix_operations.py

from typing import List, Any

# Global cache to store computation results
computation_cache = {}

def matrix_operations(matrix_a: List[List[float]], matrix_b: List[List[float]]) -> List[List[float]]:
    """
    Multiplies two matrices and stores the result in the global computation_cache.

    Args:
        matrix_a (List[List[float]]): The first matrix.
        matrix_b (List[List[float]]): The second matrix.

    Returns:
        List[List[float]]: The result of matrix multiplication.

    Raises:
        ValueError: If the matrices cannot be multiplied due to incompatible dimensions.
    """
    # Validate input matrices
    if not matrix_a or not matrix_b:
        raise ValueError("Input matrices must not be empty.")

    num_rows_a = len(matrix_a)
    num_cols_a = len(matrix_a[0])
    num_rows_b = len(matrix_b)
    num_cols_b = len(matrix_b[0])

    # Check if all rows in matrix_a have the same number of columns
    for row in matrix_a:
        if len(row) != num_cols_a:
            raise ValueError("All rows in matrix_a must have the same number of columns.")

    # Check if all rows in matrix_b have the same number of columns
    for row in matrix_b:
        if len(row) != num_cols_b:
            raise ValueError("All rows in matrix_b must have the same number of columns.")

    # Check if matrices can be multiplied
    if num_cols_a != num_rows_b:
        raise ValueError("Number of columns in matrix_a must equal number of rows in matrix_b.")

    # Matrix multiplication
    result = []
    for i in range(num_rows_a):
        result_row = []
        for j in range(num_cols_b):
            sum_product = 0.0
            for k in range(num_cols_a):
                sum_product += matrix_a[i][k] * matrix_b[k][j]
            result_row.append(sum_product)
        result.append(result_row)

    # Store result in global computation_cache
    # Use tuple of tuples as keys for hashability
    key = (tuple(tuple(row) for row in matrix_a), tuple(tuple(row) for row in matrix_b))
    computation_cache[key] = result

    return result
